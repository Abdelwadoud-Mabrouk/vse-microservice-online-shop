\section{Geschäftsprozess und technische Umsetzung}

\begin{frame}{Bestellprozess im Überblick}
    \begin{columns}[T]
        \begin{column}{.52\textwidth}
            \begin{block}{Ablauf}
                \begin{enumerate}
                    \item Produkt-IDs $\rightarrow$ Shop $\rightarrow$ \texttt{CREATED}
                    \item Warehouse reserviert Bestand
                    \item Total berechnen $\rightarrow$ \texttt{PROCESSING}
                    \item Billing startet Zahlung
                    \item Kunde erhält URL $\rightarrow$ \texttt{AWAITING\_PAYMENT}
                \end{enumerate}
            \end{block}
        \end{column}
        \begin{column}{.44\textwidth}
            \begin{alertblock}{Orchestration-based Saga}
                \begin{itemize}
                    \item Fehler $\rightarrow$ Kompensation
                    \item Order bleibt mit \texttt{CANCELLED}
                    \item Audit protokolliert alle Schritte
                \end{itemize}
            \end{alertblock}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{Zahlungsabwicklung und Provider-Integration}
\begin{itemize}
    \item Billing Service verarbeitet den gesamten Zahlungsprozess
    \item Integration von Stripe und PayPal über ein gemeinsames Interface
    \item Plugin Design ermöglicht austauschbare Provider
    \item Keine Änderungen an der Geschäftslogik bei neuen Anbietern nötig
\end{itemize}

\vspace{.3cm}

\begin{block}{Kernidee}
Die Zahlungslogik bleibt unabhängig vom konkreten Anbieter.
\end{block}

\end{frame}

\begin{frame}{Speicherung und Nachvollziehbarkeit von Zahlungen}
\begin{itemize}
    \item Provider-Antworten werden im Billing Service gespeichert
    \item Enthält Status-, Referenz- und Transaktionsdaten
    \item Ereignisse werden zusätzlich über Kafka an den Audit Service gesendet
    \item Trennung zwischen Geschäftsdaten und Audit-Historie
\end{itemize}

\vspace{.3cm}

\begin{columns}[T]
    \begin{column}{.48\textwidth}
        \begin{block}{Billing Datenbank}
            Operative Daten für die Zahlungsabwicklung
        \end{block}
    \end{column}
    \begin{column}{.48\textwidth}
        \begin{block}{Audit Datenbank}
            Vollständige Historie aller relevanten Ereignisse
        \end{block}
    \end{column}
\end{columns}
\end{frame}

\begin{frame}{Webhook und Event-Verarbeitung}
    \begin{columns}[T]
        \begin{column}{.48\textwidth}
            \begin{block}{Billing Service – Producer}
                \begin{itemize}
                    \item Empfängt Webhooks vom Provider
                    \item Aktualisiert den Zahlungsstatus
                    \item Publiziert Events über Kafka
                \end{itemize}
            \end{block}
        \end{column}
        \begin{column}{.48\textwidth}
            \begin{block}{Shop Service – Consumer}
                \begin{itemize}
                    \item \textbf{SUCCESS}: Reservierung committen
                    \item \textbf{CANCELED}: Reservierung zurücknehmen
                    \item Audit protokolliert jeden Prozessschritt
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{Datenhaltung und Nachvollziehbarkeit}
    \begin{columns}[T]
        \begin{column}{.48\textwidth}
            \begin{block}{PostgreSQL}
                \begin{itemize}
                    \item Strukturierte relationale Daten
                    \item Klare Beziehungen zwischen Entitäten
                    \item Gute Konsistenz und Nachvollziehbarkeit
                \end{itemize}
            \end{block}
        \end{column}
        \begin{column}{.48\textwidth}
            \begin{block}{Monitoring und Audit}
                \begin{itemize}
                    \item Correlation-ID pro Bestellung
                    \item Audit speichert alle Schritte append-only
                    \item Grafana und Loki für Monitoring und Analyse
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
\end{frame}
