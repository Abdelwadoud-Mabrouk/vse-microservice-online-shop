\section{Technologie}

\begin{frame}{Warum .NET 10 für unsere Services?}
    \begin{itemize}
        \item Wir setzen Warehouse, Invoice und Audit mit .NET 10 um.
        \item .NET passt gut für schlanke REST-APIs.
        \item Swagger, Dependency Injection und Tests sind direkt gut unterstützt.
        \item Die Sprache C\# erlaubt klaren, gut lesbaren Code.
    \end{itemize}

    \vspace{.5cm}
    \begin{block}{Entscheidung}
        .NET 10 ist für unseren Teil pragmatisch: modern, stabil und gut testbar.
    \end{block}
\end{frame}

\begin{frame}{Warum Clean Architecture?}
    \begin{itemize}
        \item Controller enthalten keine Businesslogik.
        \item Use Cases beschreiben die fachlichen Abläufe.
        \item Domain enthält die Regeln.
        \item Infrastructure enthält Technik wie Datenbank, HTTP und Dateien.
    \end{itemize}

    \vspace{.5cm}
    \begin{block}{Vorteil}
        Wir können fachliche Logik testen, ohne echte Datenbank oder echte HTTP-Aufrufe zu brauchen.
    \end{block}
\end{frame}

\section{Kommunikation}

\begin{frame}{REST und Kafka}
    \begin{columns}[T]
        \begin{column}{.48\textwidth}
            \begin{block}{REST}
                \begin{itemize}
                    \item direkte Anfrage
                    \item direkte Antwort
                    \item gut für Lager und Zahlung
                \end{itemize}
            \end{block}
        \end{column}
        \begin{column}{.48\textwidth}
            \begin{block}{Kafka}
                \begin{itemize}
                    \item asynchrone Events
                    \item Services sind weniger gekoppelt
                    \item gut für Rechnung
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}

    \vspace{.5cm}
    \begin{alertblock}{Warum beides?}
        REST ist gut, wenn der Shop sofort eine Antwort braucht.
        Kafka ist gut, wenn ein Schritt später im Hintergrund passieren kann.
    \end{alertblock}
\end{frame}

\begin{frame}{Warum Kafka?}
    \begin{itemize}
        \item Kafka ist sinnvoll für Ereignisse, die nicht sofort beantwortet werden müssen.
        \item Beispiel: Nach erfolgreicher Zahlung entsteht eine Rechnung.
        \item Invoice-Service kann unabhängig vom Shop arbeiten.
        \item Später können weitere Services dasselbe Event nutzen.
    \end{itemize}

    \vspace{.5cm}
    \begin{block}{Beispiel}
        Billing sendet \texttt{PaymentSucceeded}. Invoice hört darauf und erstellt die Rechnung.
    \end{block}
\end{frame}

\section{Datenhaltung}

\begin{frame}{Warum PostgreSQL für Audit?}
    \begin{itemize}
        \item Audit-Daten müssen zuverlässig gespeichert werden.
        \item PostgreSQL ist stabil und gut geeignet für strukturierte Daten.
        \item Die Timeline einer Bestellung kann einfach abgefragt werden.
        \item Append-only Speicherung passt gut zum Audit-Prinzip.
    \end{itemize}

    \vspace{.5cm}
    \begin{block}{Entscheidung}
        Für Audit ist Konsistenz wichtiger als maximale Geschwindigkeit.
        Deshalb verwenden wir PostgreSQL.
    \end{block}
\end{frame}

\begin{frame}{Audit: Warum ist das wichtig?}
    \begin{block}{Ziel}
        Wir können später genau sehen, was mit einer Bestellung passiert ist.
    \end{block}

    \begin{itemize}
        \item Jede Bestellung bekommt eine \texttt{correlationId}.
        \item Alle Services schreiben Events mit dieser ID.
        \item Dadurch entsteht eine Timeline der Bestellung.
        \item Audit-Einträge werden nicht verändert.
    \end{itemize}
\end{frame}

            \begin{block}{Best Case}

                    \item Rechnung erstellt
                    \item Bestellung abgeschlossen
                \end{itemize}
            \end{block}
        \end{column}
        \begin{column}{.48\textwidth}
            \begin{alertblock}{Worst Case}
                \begin{itemize}
                    \item Zahlung erfolgreich
                    \item Warehouse-Commit schlägt fehl
                    \item Refund wird ausgelöst
                    \item Audit zeigt Fehler und Ausgleich
                \end{itemize}
            \end{alertblock}
        \end{column}
    \end{columns}

    \vspace{.45cm}
    \begin{block}{Warum wichtig?}
        Die Architektur muss nicht nur im Erfolgsfall funktionieren, sondern auch bei Fehlern nachvollziehbar bleiben.
    \end{block}
\end{frame}
