\section{Technologie}

\begin{frame}{Warum Clean Architecture?}
    \begin{itemize}
        \item Die API-Schicht nimmt nur Anfragen an.
        \item Use Cases beschreiben die fachlichen Abläufe.
        \item Domain enthält Modelle und Regeln.
        \item Infrastructure enthält Datenbank, Kafka, HTTP und Dateien.
    \end{itemize}

    \vspace{.5cm}
    \begin{block}{Vorteil}
        Die Businesslogik bleibt unabhängig von Frameworks, Datenbank und UI.
    \end{block}
\end{frame}

\section{Kommunikation}

\begin{frame}{REST und Kafka}
    \begin{columns}[T]
        \begin{column}{.48\textwidth}
            \begin{block}{REST}
                \begin{itemize}
                    \item direkte Anfrage
                    \item direkte Antwort
                    \item gut für Lager-Reservierung
                \end{itemize}
            \end{block}
        \end{column}
        \begin{column}{.48\textwidth}
            \begin{block}{Kafka}
                \begin{itemize}
                    \item asynchrone Events
                    \item Services sind weniger gekoppelt
                    \item gut für Rechnung und Audit
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}

    \vspace{.5cm}
    \begin{alertblock}{Warum beides?}
        REST nutzen wir, wenn der Shop sofort wissen muss, ob etwas geklappt hat.
        Kafka nutzen wir, wenn ein Ereignis im Hintergrund weiterverarbeitet wird.
    \end{alertblock}
\end{frame}

\section{Datenhaltung}

\begin{frame}{Datenhaltung}
    \begin{itemize}
        \item Jeder Service besitzt seine eigenen Daten.
        \item Shop speichert Bestellungen und Preise.
        \item Warehouse speichert Bestand und Reservierungen.
        \item Invoice speichert Rechnungsdaten und PDF-Dateien.
        \item Audit speichert alle Events append-only in PostgreSQL.
    \end{itemize}

    \vspace{.5cm}
    \begin{block}{Wichtig}
        Ein Service greift nicht direkt auf die Datenbank eines anderen Services zu.
    \end{block}
\end{frame}

\begin{frame}{Audit und Observability}
    \begin{block}{Ziel}
        Wir können genau sehen, was mit einer Bestellung passiert ist.
    \end{block}

    \begin{itemize}
        \item Jede Bestellung bekommt eine \texttt{correlationId}.
        \item Audit zeigt die fachliche Timeline.
        \item Grafana zeigt technische Logs für Debugging.
        \item Beide helfen, Fehler schneller zu finden.
    \end{itemize}
\end{frame}

\section{Abläufe}

\begin{frame}{Geplanter Best Case und Worst Case}
    \begin{columns}[T]
        \begin{column}{.48\textwidth}
            \begin{block}{Best Case}
                \begin{itemize}
                    \item Bestellung wird angelegt
                    \item Lager reserviert
                    \item Zahlung wird simuliert
                    \item Rechnung erstellt
                    \item Lager wird committed
                    \item Bestellung abgeschlossen
                \end{itemize}
            \end{block}
        \end{column}
        \begin{column}{.48\textwidth}
            \begin{alertblock}{Worst Case}
                \begin{itemize}
                    \item Bestellung wird angelegt
                    \item Bestand reicht nicht aus
                    \item Warehouse liefert \texttt{409 Conflict}
                    \item Keine Rechnung wird erstellt
                    \item Audit zeigt den Fehler
                \end{itemize}
            \end{alertblock}
        \end{column}
    \end{columns}

    \vspace{.45cm}
    \begin{block}{Warum wichtig?}
        Die Architektur muss nicht nur im Erfolgsfall funktionieren.
        Sie muss auch erklären können, warum ein Prozess gestoppt wurde.
    \end{block}
\end{frame}

\begin{frame}{Zusammenfassung}
    \begin{itemize}
        \item Microservices sind die Vorgabe der Aufgabe.
        \item Unsere Entscheidungen sind: Java Spring Boot, .NET 10, Clean Architecture, REST, Kafka und PostgreSQL.
        \item REST nutzen wir für direkte Entscheidungen.
        \item Kafka nutzen wir für entkoppelte Events und Audit.
        \item PostgreSQL nutzen wir für strukturierte, nachvollziehbare Daten.
        \item Die \texttt{correlationId} verbindet die Schritte einer Bestellung.
    \end{itemize}

\end{frame}
