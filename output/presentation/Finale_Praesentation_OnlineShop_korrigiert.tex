\centering
\begin{tikzpicture}[node distance=.75cm]

  \node[bad,minimum width=2.4cm,right=2.8cm of r,yshift=-.8cm] (x) {\textbf{CANCEL}\\wieder freigeben};
  \draw[arrow] (r.east)--(c.west) node[pos=.56,above=2pt,sloped,font=\scriptsize]{Zahlung erfolgreich};
  \draw[arrow] (r.east)--(x.west) node[pos=.56,below=2pt,sloped,font=\scriptsize]{Abbruch / Fehler};
\end{tikzpicture}

\vspace{.65cm}
\begin{itemize}
  \item Reservierung trennt „verfügbar“ von „endgültig verkauft“.
  \item \texttt{orderId} identifiziert genau eine Reservierung.
  \item Wiederholtes Commit oder Cancel verändert den Bestand nicht erneut.
  \item Kritische Bestandsänderungen werden transaktional abgesichert.
\end{itemize}
\end{frame}
